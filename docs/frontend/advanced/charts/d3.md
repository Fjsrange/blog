# D3.js 数据可视化

D3.js（Data-Driven Documents）是一个强大的 JavaScript 可视化库，通过数据驱动 DOM 操作，实现高度定制化的数据可视化。

## 核心思想

```
数据 → 变换 → DOM/SVG/Canvas

D3 的核心是数据绑定：
将数据数组与 DOM 元素集合建立对应关系，
通过 enter/update/exit 模式管理元素的生命周期。
```

## 安装与引入

```bash
npm install d3
```

```javascript
// 完整引入
import * as d3 from 'd3';

// 按需引入
import { select, scaleLinear, line, axisBottom } from 'd3';
```

## 选择与操作

```javascript
// 选择元素
const svg = d3.select('#chart');        // 选择第一个匹配
const items = d3.selectAll('.item');    // 选择所有匹配

// 链式操作
d3.select('#chart')
  .append('svg')
  .attr('width', 600)
  .attr('height', 400)
  .style('background', '#f5f5f5');

// 属性操作
svg.attr('width', 600);          // 设置属性
svg.style('fill', 'steelblue');  // 设置样式
svg.classed('active', true);     // 设置 class
svg.text('Hello D3');            // 设置文本
svg.html('<rect></rect>');       // 设置 HTML

// 基于数据动态设置
svg.selectAll('circle')
  .data(dataset)
  .attr('cx', d => xScale(d.x))
  .attr('cy', d => yScale(d.y))
  .attr('r', d => Math.sqrt(d.value));
```

## 数据绑定

```javascript
// enter / update / exit 模式
const circles = svg.selectAll('circle')
  .data(dataset, d => d.id); // key 函数确保绑定稳定

// 更新已有元素
circles
  .attr('cx', d => xScale(d.x))
  .attr('cy', d => yScale(d.y))
  .attr('fill', 'steelblue');

// 创建新元素
circles.enter()
  .append('circle')
  .attr('cx', d => xScale(d.x))
  .attr('cy', d => yScale(d.y))
  .attr('r', 0)
  .transition()
  .duration(500)
  .attr('r', 5)
  .attr('fill', 'steelblue');

// 移除多余元素
circles.exit()
  .transition()
  .duration(500)
  .attr('r', 0)
  .remove();

// 简化写法（D3 v5+）
svg.selectAll('circle')
  .data(dataset)
  .join(
    enter => enter.append('circle')
      .attr('cx', d => xScale(d.x))
      .attr('cy', d => yScale(d.y))
      .attr('r', 0)
      .call(enter => enter.transition().attr('r', 5)),
    update => update
      .call(update => update.transition()
        .attr('cx', d => xScale(d.x))
        .attr('cy', d => yScale(d.y))),
    exit => exit
      .call(exit => exit.transition().attr('r', 0).remove()),
  );
```

## 比例尺

```javascript
// 线性比例尺
const linear = d3.scaleLinear()
  .domain([0, 100])    // 数据范围
  .range([0, 500])     // 像素范围
  .clamp(true);        // 限制输出范围
linear(50); // 250

// 序数比例尺
const ordinal = d3.scaleOrdinal()
  .domain(['A', 'B', 'C'])
  .range(['#5470c6', '#91cc75', '#fac858']);
ordinal('A'); // '#5470c6'

// 颜色比例尺
const color = d3.scaleSequential(d3.interpolateViridis)
  .domain([0, 100]);

// 时间比例尺
const time = d3.scaleTime()
  .domain([new Date(2024, 0, 1), new Date(2024, 11, 31)])
  .range([0, 800]);

// 带宽比例尺（柱状图）
const band = d3.scaleBand()
  .domain(['A', 'B', 'C', 'D'])
  .range([0, 400])
  .padding(0.2);
band('A');        // 起始位置
band.bandwidth(); // 每个带宽的宽度
```

## 坐标轴

```javascript
// 创建坐标轴
const xAxis = d3.axisBottom(xScale)
  .ticks(5)                    // 刻度数量
  .tickFormat(d3.format('.0f')); // 格式化

const yAxis = d3.axisLeft(yScale)
  .ticks(5)
  .tickFormat(d => `${d}%`);

// 渲染坐标轴
svg.append('g')
  .attr('class', 'x-axis')
  .attr('transform', `translate(0, ${height})`)
  .call(xAxis);

svg.append('g')
  .attr('class', 'y-axis')
  .call(yAxis);

// 更新坐标轴
svg.select('.x-axis')
  .transition()
  .duration(500)
  .call(xAxis);
```

## 绘制图表

### 柱状图

```javascript
function barChart(data) {
  const width = 600, height = 400;
  const margin = { top: 20, right: 20, bottom: 40, left: 50 };

  const x = d3.scaleBand()
    .domain(data.map(d => d.name))
    .range([margin.left, width - margin.right])
    .padding(0.2);

  const y = d3.scaleLinear()
    .domain([0, d3.max(data, d => d.value)]).nice()
    .range([height - margin.bottom, margin.top]);

  const svg = d3.select('#chart')
    .append('svg')
    .attr('width', width)
    .attr('height', height);

  svg.append('g')
    .attr('transform', `translate(0,${height - margin.bottom})`)
    .call(d3.axisBottom(x));

  svg.append('g')
    .attr('transform', `translate(${margin.left},0)`)
    .call(d3.axisLeft(y));

  svg.selectAll('.bar')
    .data(data)
    .join('rect')
    .attr('class', 'bar')
    .attr('x', d => x(d.name))
    .attr('y', height - margin.bottom)
    .attr('width', x.bandwidth())
    .attr('height', 0)
    .attr('fill', 'steelblue')
    .transition()
    .duration(500)
    .attr('y', d => y(d.value))
    .attr('height', d => height - margin.bottom - y(d.value));
}
```

### 折线图

```javascript
function lineChart(data) {
  const width = 600, height = 400;
  const margin = { top: 20, right: 20, bottom: 40, left: 50 };

  const x = d3.scaleTime()
    .domain(d3.extent(data, d => d.date))
    .range([margin.left, width - margin.right]);

  const y = d3.scaleLinear()
    .domain(d3.extent(data, d => d.value)).nice()
    .range([height - margin.bottom, margin.top]);

  const line = d3.line()
    .x(d => x(d.date))
    .y(d => y(d.value))
    .curve(d3.curveMonotoneX); // 平滑曲线

  const svg = d3.select('#chart')
    .append('svg')
    .attr('width', width)
    .attr('height', height);

  svg.append('path')
    .datum(data)
    .attr('fill', 'none')
    .attr('stroke', 'steelblue')
    .attr('stroke-width', 2)
    .attr('d', line);
}
```

```javascript
svg.selectAll('circle')

  .join('circle')
  .on('mouseover', function(event, d) {
    d3.select(this)
      .transition()
      .attr('r', 8)

    tooltip
      .style('opacity', 1)
      .html(`<strong>${d.name}</strong><br>值: ${d.value}`)
      .style('left', `${event.pageX + 10}px`)
      .style('top', `${event.pageY - 20}px`);
  })
  .on('mouseout', function() {
    d3.select(this)
      .transition()
      .attr('r', 5)
      .attr('fill', 'steelblue');

    tooltip.style('opacity', 0);
  });

// 缩放
const zoom = d3.zoom()
  .scaleExtent([0.5, 5])
  .on('zoom', (event) => {
    svg.attr('transform', event.transform);
  });

svg.call(zoom);

// 拖拽
const drag = d3.drag()
  .on('start', function(event, d) {
    d3.select(this).raise().attr('stroke', 'black');
  })
  .on('drag', function(event, d) {
    d3.select(this)
      .attr('cx', d.x = event.x)
      .attr('cy', d.y = event.y);
  })
  .on('end', function(event, d) {
    d3.select(this).attr('stroke', null);
  });

svg.selectAll('circle').call(drag);
```
